

f=$(find /www/server/panel/vhost/cert -name "fullchain.pem" 2>/dev/null | head -1)
echo "证书: $f"
/www/server/panel/pyenv/bin/python - "$f" <<'EOF'
import sys
from OpenSSL import crypto
from cryptography import x509 as cx

c = crypto.load_certificate(crypto.FILETYPE_PEM, open(sys.argv[1], 'rb').read())
b = c.to_cryptography()
try:
    san = b.extensions.get_extension_for_class(cx.SubjectAlternativeName)
    dns = [str(g.value) for g in san.value]
    print("dns =", dns)
    print("数量 =", len(dns))
except cx.ExtensionNotFound:
    print("该证书没有 SAN")
EOF
